local function setStage(stage, detail)

local function logError(tag, err)
    STATE.Errors[tag] = tostring(err)
    warn("[AXIOM KAITUN][" .. tostring(tag) .. "]", err)
end

local function safe(tag, fn)
    local ok, result = pcall(fn)

    if not ok then
        logError(tag, result)
        return false, result
    end

    return true, result
end

local function getCommF()
    local remotes = ReplicatedStorage:FindFirstChild("Remotes")

    if not remotes then
        return nil
    end

    return remotes:FindFirstChild("CommF_")
end

local function invoke(...)
    local remote = getCommF()

    if not remote then
        return false, "NO_COMMF"
    end

    local args = table.pack(...)

    return safe("CommF", function()
        return remote:InvokeServer(
            table.unpack(args, 1, args.n)
        )
    end)
end

local function getCharacter()
    local character = Player.Character

    if not character then
        return nil
    end

    local humanoid =
        character:FindFirstChildOfClass("Humanoid")

    local root =
        character:FindFirstChild("HumanoidRootPart")

    if not humanoid or not root or humanoid.Health <= 0 then
        return nil
    end

    return character, humanoid, root
end

local function getDataValue(name, default)
    local data = Player:FindFirstChild("Data")
    local value = data and data:FindFirstChild(name)

    if value then
        return value.Value
    end

    return default
end

local function getLevel()
    return tonumber(getDataValue("Level", 1)) or 1
end

local function getBeli()
    return tonumber(getDataValue("Beli", 0)) or 0
end

local function getFragments()
    return tonumber(getDataValue("Fragments", 0)) or 0
end

local function getPoints()
    return tonumber(getDataValue("Points", 0)) or 0
end

local function getSea()
    if game.PlaceId == PLACE_SEA1 then
        return 1
    elseif game.PlaceId == PLACE_SEA2 then
        return 2
    elseif game.PlaceId == PLACE_SEA3 then
        return 3
    end

    return 0
end

local function chooseTeam()
    local main = PlayerGui:FindFirstChild("Main")

    if not main then
        return
    end

    local choose = main:FindFirstChild("ChooseTeam", true)

    if not choose or not choose.Visible then
        return
    end

    local wanted =
        string.lower(CONFIG.Team or "Pirates")

    for _, obj in ipairs(choose:GetDescendants()) do
        if obj:IsA("TextButton") then
            local text =
                string.lower(
                    tostring(obj.Text or obj.Name)
                )

            if string.find(text, wanted, 1, true) then
                safe("ChooseTeam", function()
                    obj:Activate()
                end)

                return
            end
        end
    end
end

task.spawn(function()
    while ScreenGui.Parent do
        safe("Team", chooseTeam)
        task.wait(2)
    end
end)

local function resolveQuest(level)
    for _, q in ipairs(QUESTS) do
        if level >= q[1] and level <= q[2] then
            return {
                Min = q[1],
                Max = q[2],
                Mob = q[3],
                Quest = q[4],
                Index = q[5],
                CFrame = q[6],
            }
        end
    end

    return nil
end

    local main = PlayerGui:FindFirstChild("Main")
    return main and main:FindFirstChild("Quest")
end

    local quest = getQuestGui()

    if not quest then
        return false
    end

    local ok, visible = pcall(function()
        return quest.Visible
    end)

    return ok and visible
end

local function questText()
    local quest = getQuestGui()

    if not quest then
        return ""
    end

    local texts = {}

    for _, obj in ipairs(quest:GetDescendants()) do
        if obj:IsA("TextLabel")
            or obj:IsA("TextButton") then

            if obj.Text and obj.Text ~= "" then
                table.insert(texts, obj.Text)
            end
        end
    end

    return table.concat(texts, " ")
end

local function setMovementOwner(owner)
    if STATE.MovementOwner
        and STATE.MovementOwner ~= owner then

        return false
    end

    STATE.MovementOwner = owner
    return true
end

local function releaseMovement(owner)
    if STATE.MovementOwner == owner then
        STATE.MovementOwner = nil
    end
end

local function flatCFrame(position, source)
    local look =
        source and source.LookVector
        or Vector3.new(0, 0, -1)

    local flat =
        Vector3.new(
            look.X,
            0,
            look.Z
        )

    if flat.Magnitude < 0.01 then
        flat = Vector3.new(0, 0, -1)
    else
        flat = flat.Unit
    end

    return CFrame.lookAt(
        position,
        position + flat
    )
end

local function tweenPhase(root, cf, speed, token)
    if not root
        or not root.Parent
        or token ~= STATE.TravelToken then

        return false
    end

    local distance =
        (root.Position - cf.Position).Magnitude

    if distance <= 3 then
        root.CFrame = cf
        return true
    end

    local tween =
        TweenService:Create(
            root,
            TweenInfo.new(
                math.clamp(
                    distance / speed,
                    0.08,
                    30
                ),
                Enum.EasingStyle.Linear
            ),
            {CFrame = cf}
        )

    tween:Play()

    local state = tween.Completed:Wait()

    return token == STATE.TravelToken
        and state == Enum.PlaybackState.Completed
end

local function travelTo(owner, destination)
    local character, humanoid, root =
        getCharacter()

    if not root then
        return false
    end

    if not setMovementOwner(owner) then
        return false
    end

    STATE.TravelToken += 1
    local token = STATE.TravelToken

    pcall(function()
        humanoid.Sit = false
        humanoid.AutoRotate = false
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
    end)

    local start = root.Position
    local target = destination.Position

    local horizontal =
        (
            Vector2.new(start.X, start.Z)
            - Vector2.new(target.X, target.Z)
        ).Magnitude

    local ok = true

    if horizontal > 450 then
        local safeY =
            math.max(
                CONFIG.SafeHeight,
                start.Y + 120,
                target.Y + 120
            )

        ok = tweenPhase(
            root,
            flatCFrame(
                Vector3.new(start.X, safeY, start.Z),
                root.CFrame
            ),
            CONFIG.VerticalSpeed,
            token
        )

        if ok then
            ok = tweenPhase(
                root,
                flatCFrame(
                    Vector3.new(target.X, safeY, target.Z),
                    root.CFrame
                ),
                CONFIG.TravelSpeed,
                token
            )
        end

        if ok then
            ok = tweenPhase(
                root,
                flatCFrame(target, destination),
                CONFIG.VerticalSpeed,
                token
            )
        end
    else
        ok = tweenPhase(
            root,
            flatCFrame(target, destination),
            CONFIG.TravelSpeed,
            token
        )
    end

    pcall(function()
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
        humanoid.AutoRotate = true
    end)

    releaseMovement(owner)

    return ok
end

local function cancelTravel()

    STATE.MovementOwner = nil
end

local function equipBestTool()
    local character, humanoid =
        getCharacter()

    if not character or not humanoid then
        return nil
    end

    local backpack =
        Player:FindFirstChild("Backpack")

    local preferred = {
        "Sanguine Art",
        "Godhuman",
        "Electric Claw",
        "Sharkman Karate",
        "Death Step",
        "Dragon Talon",
        "Superhuman",
        "Dragon Claw",
        "Fishman Karate",
        "Electro",
        "Black Leg",
        "Combat",
    }

    for _, name in ipairs(preferred) do
        local tool =
            character:FindFirstChild(name)
            or (backpack and backpack:FindFirstChild(name))

        if tool and tool:IsA("Tool") then
            if tool.Parent == backpack then
                humanoid:EquipTool(tool)
                task.wait()
            end

            return tool
        end
    end

    if backpack then
        for _, tool in ipairs(backpack:GetChildren()) do
            if tool:IsA("Tool")
                and not tool:FindFirstChild("Fruit") then

                humanoid:EquipTool(tool)
                task.wait()
                return tool
            end
        end
    end

    return character:FindFirstChildOfClass("Tool")
end

local NetAttack = nil
local NetHit = nil

local function initNet()
    if NetAttack and NetHit then
        return true
    end

    local modules =
        ReplicatedStorage:FindFirstChild("Modules")

    local net =
        modules and modules:FindFirstChild("Net")

    if not net then
        return false
    end

    NetAttack =
        net:FindFirstChild("RE/RegisterAttack")

    NetHit =
        net:FindFirstChild("RE/RegisterHit")

    return NetAttack ~= nil
        and NetHit ~= nil
end

local function getHitPart(enemy)
    return enemy:FindFirstChild("Head")
        or enemy:FindFirstChild("HumanoidRootPart")
        or enemy:FindFirstChildWhichIsA("BasePart")
end

local function enemyMatches(enemy, wanted)
    if not enemy
        or not enemy:IsA("Model") then

        return false
    end

    return string.find(
        string.lower(enemy.Name),
        string.lower(wanted),
        1,
        true
    ) ~= nil
end

local function getEnemiesFolder()
    return Workspace:FindFirstChild("Enemies")
end

local function getTargets(wanted, origin)
    local result = {}
    local folder = getEnemiesFolder()

    if not folder then
        return result
    end

    for _, enemy in ipairs(folder:GetChildren()) do
        if enemyMatches(enemy, wanted) then
            local hum =
                enemy:FindFirstChildOfClass("Humanoid")

            local root =
                enemy:FindFirstChild("HumanoidRootPart")

            if hum
                and root
                and hum.Health > 0 then

                table.insert(result, {
                    Enemy = enemy,
                    Humanoid = hum,
                    Root = root,
                    Distance =
                        (root.Position - origin).Magnitude,
                })
            end
        end
    end

    table.sort(result, function(a, b)
        return a.Distance < b.Distance
    end)

    return result
end

local function collectCombatTargets(targets, root)
    local others = {}
    local basePart = nil

    if not root or type(targets) ~= "table" then
        return basePart, others
    end

    for _, data in ipairs(targets) do
        local enemy = data.Enemy
        local humanoid = data.Humanoid

        if enemy
            and enemy.Parent
            and humanoid
            and humanoid.Health > 0 then

            -- Current fast-attack sources prefer Head.
            local head =
                enemy:FindFirstChild("Head")
                or enemy:FindFirstChild("HumanoidRootPart")

            if head and head:IsA("BasePart") then
                local distance =
                    (head.Position - root.Position).Magnitude

                if distance <= CONFIG.AttackRange then
                    table.insert(
                        others,
                        {
                            enemy,
                            head,
                        }
                    )

                    -- Same shape used by RegisterHit:
                    -- BasePart + {{Enemy, Head}, ...}
                    basePart = head

                    if #others >= 8 then
                        break
                    end
                end
            end
        end
    end

    return basePart, others
end

local function attack(tool, targets, root)
    if not tool
        or not tool.Parent
        or not root
        or os.clock() - STATE.LastAttack
            < CONFIG.AttackDelay then

        return false, "COOLDOWN"
    end

    -- Tool phải đang nằm trong Character mới đánh ổn định.
    local character = Player.Character

    if not character then
        return false, "NO_CHARACTER"
    end

    if tool.Parent ~= character then
        local humanoid =
            character:FindFirstChildOfClass("Humanoid")

        if humanoid then
            safe("EquipAttackTool", function()
                humanoid:EquipTool(tool)
            end)

            task.wait()
        end
    end

    if tool.Parent ~= character then
        return false, "TOOL_NOT_EQUIPPED"
    end

    local basePart, others =
        collectCombatTargets(
            targets,
            root
        )

    if not basePart or #others == 0 then
        return false, "NO_TARGET_IN_60"
    end

    STATE.LastAttack = os.clock()

    local usedLeft = false
    local usedNet = false
    local usedActivate = false

    --=====================================================
    -- BACKEND 1: LeftClickRemote
    -- Current source loops every target in range.
    --=====================================================

    local left =
        tool:FindFirstChild("LeftClickRemote")
        or tool:FindFirstChild(
            "LeftClickRemote",
            true
        )

    if left and left:IsA("RemoteEvent") then
        local pivotPosition =
            character:GetPivot().Position

        for _, pair in ipairs(others) do
            local enemy = pair[1]
            local enemyRoot =
                enemy
                and enemy:FindFirstChild("HumanoidRootPart")

            if enemyRoot then
                local delta =
                    enemyRoot.Position
                    - pivotPosition

                if delta.Magnitude > 0.01 then
                    local ok = pcall(function()
                        left:FireServer(
                            delta.Unit,
                            1
                        )
                    end)

                    if ok then
                        usedLeft = true
                    end
                end
            end
        end
    end

    --=====================================================
    -- BACKEND 2: Modules.Net RegisterAttack/RegisterHit
    -- Current source:
    --   RegisterAttack:FireServer(1e-9)
    --   RegisterHit:FireServer(BasePart, OthersEnemies)
    --
    -- We also run this after LeftClickRemote as a reliability
    -- fallback because different fighting styles/tools differ.
    --=====================================================

    if initNet() then
        local attackOK = pcall(function()
            NetAttack:FireServer(1e-9)
        end)

        local hitOK = false

        if attackOK then
            hitOK = pcall(function()
                NetHit:FireServer(
                    basePart,
                    others
                )
            end)
        end

        usedNet =
            attackOK and hitOK
    end

    --=====================================================
    -- BACKEND 3: normal Tool activation fallback
    --=====================================================

    if not usedLeft and not usedNet then
        local ok = pcall(function()
            tool:Activate()
        end)

        usedActivate = ok
    end

    local backend = {}

    if usedLeft then
        table.insert(
            backend,
            "LEFT"
        )
    end

    if usedNet then
        table.insert(
            backend,
            "NET"
        )
    end

    if usedActivate then
        table.insert(
            backend,
            "ACTIVATE"
        )
    end

    if #backend == 0 then
        return false,
            "ALL_ATTACK_BACKENDS_FAILED"
    end

    return true,
        table.concat(
            backend,
            "+"
        )
end

local function hoverTarget(target)
    local character, humanoid, root =
        getCharacter()

    if not root
        or not target
        or not target.Root then

        return
    end

    if STATE.MovementOwner then
        return
    end

    local p = target.Root.Position

    local farmPos =
        Vector3.new(
            p.X,
            p.Y + CONFIG.FarmHeight,
            p.Z
        )

    local look =
        Vector3.new(
            p.X,
            farmPos.Y,
            p.Z
        )

    pcall(function()
        humanoid.AutoRotate = false
        humanoid.Sit = false

        root.AssemblyLinearVelocity =
            Vector3.zero

        root.AssemblyAngularVelocity =
            Vector3.zero

        root.CFrame =
            CFrame.lookAt(
                farmPos,
                look
            )
    end)
end

                    CONFIG.StatBatch,

                addStat(stat, amount)
                return
            end
        end
    end

    return tool:FindFirstChild("Fruit") ~= nil
        or string.find(
            string.lower(tool.Name),
            "fruit",
            1,
            true
        ) ~= nil
end

local function trackFruit(obj)
    if obj:IsA("Tool")
        and not obj:IsDescendantOf(Player)
        and isFruitTool(obj) then

        FruitCache[obj] = true
    end
end

Workspace.DescendantAdded:Connect(function(obj)
    if obj:IsA("Tool") then
        task.defer(function()
            trackFruit(obj)
        end)
    end
end)

local function randomFruit()
    if not CONFIG.AutoRandomFruit then
        return
    end

    if os.clock() - STATE.LastRandomFruit
        < CONFIG.RandomFruitDelay then

        return
    end

    STATE.LastRandomFruit = os.clock()

    invoke("Cousin", "Buy")
end

local function getFruitCatalog()
    local ok, result =
        invoke("GetFruits")

    if ok and type(result) == "table" then
        return result
    end

    return nil
end

local function normalizeFruitName(name)
    local s =
        string.lower(
            tostring(name or "")
        )

    s = s:gsub("%s+[Ff][Rr][Uu][Ii][Tt]$", "")
    s = s:gsub("[^%w]", "")

    return s
end

local function resolveFruitInternal(tool)
    local catalog = getFruitCatalog()

    if type(catalog) ~= "table" then
        return nil
    end

    local token =
        normalizeFruitName(tool.Name)

    for _, fruit in pairs(catalog) do
        local name =
            type(fruit) == "table"
            and fruit.Name
            or fruit

        if name
            and normalizeFruitName(name) == token then

            return name
        end
    end

    for _, fruit in pairs(catalog) do
        local name =
            type(fruit) == "table"
            and fruit.Name
            or fruit

        if name then
            local f =
                normalizeFruitName(name)

            if string.find(f, token, 1, true)
                or string.find(token, f, 1, true) then

                return name
            end
        end
    end

    return nil
end

local function storeOwnedFruits()
    if not CONFIG.AutoStoreFruit then
        return
    end

    local containers = {
        Player:FindFirstChild("Backpack"),
        Player.Character,
    }

    for _, container in ipairs(containers) do
        if container then
            for _, tool in ipairs(container:GetChildren()) do
                if isFruitTool(tool) then
                    local internal =
                        resolveFruitInternal(tool)

                    if internal then
                        invoke(
                            "StoreFruit",
                            internal,
                            tool
                        )

                        task.wait(0.15)
                    end
                end
            end
        end
    end
end

local function nearestWorldFruit()
    local _, _, root = getCharacter()

    if not root then
        return nil
    end

    local nearest = nil
    local nearestHandle = nil
    local nearestDistance = math.huge

    for tool in pairs(FruitCache) do
        if not tool
            or not tool.Parent
            or tool:IsDescendantOf(Player) then

            FruitCache[tool] = nil
        else
            local handle =
                tool:FindFirstChild("Handle")
                or tool:FindFirstChildWhichIsA(
                    "BasePart",
                    true
                )

            if handle then
                local d =
                    (handle.Position - root.Position).Magnitude

                if d < nearestDistance then
                    nearest = tool
                    nearestHandle = handle
                    nearestDistance = d
                end
            end
        end
    end

    return nearest, nearestHandle
end

local function pickupFruit()
    if not CONFIG.AutoPickupFruit
        or STATE.MovementOwner then

        return
    end

    local tool, handle =
        nearestWorldFruit()

    if not tool or not handle then
        return
    end

    local _, _, root =
        getCharacter()

    if not root then
        return
    end

    local old = root.CFrame

    if not setMovementOwner("FRUIT") then
        return
    end

    safe("FruitPickup", function()
        if type(firetouchinterest) == "function" then
            firetouchinterest(root, handle, 0)
            task.wait(0.05)
            firetouchinterest(root, handle, 1)
        end

        if tool.Parent
            and not tool:IsDescendantOf(Player) then

            root.CFrame =
                handle.CFrame
                * CFrame.new(0, 1.5, 0)

            task.wait(0.25)
        end

        root.CFrame = old
    end)

    releaseMovement("FRUIT")
end

task.spawn(function()
    while ScreenGui.Parent do
        if STATE.Enabled then
            safe("RandomFruit", randomFruit)
            safe("StoreFruit", storeOwnedFruits)
            safe("PickupFruit", pickupFruit)
        end

        task.wait(1)
    end
end)

local function findBoss(namePart)
    local folders = {
        Workspace:FindFirstChild("Enemies"),
        ReplicatedStorage,
    }

    for _, folder in ipairs(folders) do
        if folder then
            for _, obj in ipairs(folder:GetChildren()) do
                if obj:IsA("Model")
                    and string.find(
                        string.lower(obj.Name),
                        string.lower(namePart),
                        1,
                        true
                    ) then

                    local hum =
                        obj:FindFirstChildOfClass("Humanoid")

                    local root =
                        obj:FindFirstChild("HumanoidRootPart")

                    if hum and root and hum.Health > 0 then
                        return obj
                    end
                end
            end
        end
    end

    return nil
end

local function farmBossModel(boss, stage)
    if not boss or not boss.Parent then
        return false
    end

    local hum =
        boss:FindFirstChildOfClass("Humanoid")

    local root =
        boss:FindFirstChild("HumanoidRootPart")

    if not hum or not root then
        return false
    end

    setStage(stage, boss.Name)

    while STATE.Enabled
        and boss.Parent
        and hum.Health > 0 do

        local character, _, playerRoot =
            getCharacter()

        if not playerRoot then
            task.wait(0.2)
            continue
        end

        if (
            playerRoot.Position
            - root.Position
        ).Magnitude > 150 then

            travelTo(
                "BOSS",
                root.CFrame
                * CFrame.new(
                    0,
                    CONFIG.FarmHeight,
                    0
                )
            )
        else
            hoverTarget({
                Root = root,
            })

            local tool = equipBestTool()

            local attacked, backend =
                attack(
                    tool,
                    {
                        {
                            Enemy = boss,
                            Humanoid = hum,
                            Root = root,
                        },
                    },
                    playerRoot
                )

            if attacked then
                STATE.Detail =
                    boss.Name ..
                    " | ATK=" ..
                    tostring(backend)
            end
        end

        task.wait(0.08)
    end

    return true
end

local function autoSea()
    if not CONFIG.AutoSea then
        return false
    end

    local sea = getSea()
    local level = getLevel()

    if sea == 1 and level >= 700 then
        unlockSea2()
        return true
    end

    if sea == 2 and level >= 1500 then
        unlockSea3()
        return true
    end

    return false
end

local function farmLevel()
    local level = getLevel()

    if level >= CONFIG.MaxLevel then
        setStage(
            "MAX LEVEL",
            "Lv." .. tostring(level) ..
            " • tiếp tục Fruit/Style/Haki"
        )

        return
    end

    local q =
        resolveQuest(level)

    if not q then
        setStage(
            "QUEST ERROR",
            "Không có quest cho Lv." ..
            tostring(level)
        )

        task.wait(1)
        return
    end

    if questVisible()
        and not questMatches(q) then

        setStage(
            "QUEST",
            "Bỏ quest cũ"
        )

        abandonQuest()
        task.wait(0.25)
        return
    end

    if not questVisible() then
        startQuest(q)
        task.wait(0.25)
        return
    end

    local character, _, root =
        getCharacter()

    if not root then
        return
    end

    local targets =
        getTargets(
            q.Mob,
            root.Position
        )

    if #targets == 0 then
        local area =
            getMobArea(q)

        if area then
            setStage(
                "TRAVEL",
                "Đi bãi " .. q.Mob
            )

            specialEntrance(q)

            travelTo(
                "MOB",
                area
                * CFrame.new(
                    0,
                    CONFIG.FarmHeight,
                    0
                )
            )
        else
            setStage(
                "WAIT MOB",
                q.Mob
            )
        end

        task.wait(0.15)
        return
    end

    local target = targets[1]

    setStage(
        "FARM",
        q.Mob ..
        " | x" ..
        tostring(#targets)
    )

    local distance =
        (
            target.Root.Position
            - root.Position
        ).Magnitude

    if distance > 150 then
        travelTo(
            "TARGET",
            target.Root.CFrame
            * CFrame.new(
                0,
                CONFIG.FarmHeight,
                0
            )
        )

        return
    end

    hoverTarget(target)

    local tool =
        equipBestTool()

    local attacked, backend =
        attack(
            tool,
            targets,
            root
        )

    if attacked then
        STATE.Detail =
            q.Mob ..
            " | x" ..
            tostring(#targets) ..
            " | ATK=" ..
            tostring(backend)
    else
        STATE.Detail =
            q.Mob ..
            " | ATK WAIT=" ..
            tostring(backend)
    end
end
