local function Lower(text)
    return string.lower(tostring(text or ""))
end

local function Has(text,word)

    return string.find(
        Lower(text),
        Lower(word),
        1,
        true
    ) ~= nil
end

local function GetRoot()

    local Character =
        Player.Character
        or Player.CharacterAdded:Wait()

    return Character:FindFirstChild("HumanoidRootPart")
end

-- ============================================================
-- PROMPT INFO
-- ============================================================

local function GetPromptInfo(prompt)

    local Info =
        tostring(prompt.Name)
        .. " "
        .. tostring(prompt.ActionText)
        .. " "
        .. tostring(prompt.ObjectText)

    local Parent = prompt.Parent

    for _ = 1,6 do

        if not Parent then
            break
        end

        Info =
            Info
            .. " "
            .. tostring(Parent.Name)

        Parent = Parent.Parent
    end

    return Lower(Info)
end

-- ============================================================
-- CHECK HARVEST
-- ============================================================

local function IsHarvestPrompt(prompt)

    if not prompt:IsA("ProximityPrompt") then
        return false
    end

    if not prompt.Enabled then
        return false
    end

    local Info =
        GetPromptInfo(prompt)

    -- tránh kích hoạt các prompt không liên quan
    local Ignore = {
        "steal",
        "buy",
        "purchase",
        "sell",
        "trade",
        "gift",
        "shop",
        "open",
        "talk",
        "claim"
    }

    for _,Word in ipairs(Ignore) do
        if Has(Info,Word) then
            return false
        end
    end

    local HarvestWords = {
        "harvestprompt",
        "harvestpart",
        "harvest",
        "collect",
        "pickup",
        "pick up",
        "pick fruit",
        "pick crop",
        "fruit",
        "crop"
    }

    for _,Word in ipairs(HarvestWords) do
        if Has(Info,Word) then
            return true
        end
    end

    return false
end

-- ============================================================
-- CACHE PROMPTS
-- ============================================================

local function AddPrompt(Object)

    if Object:IsA("ProximityPrompt") then
        HarvestPrompts[Object] = true
    end
end

for _,Object in ipairs(Workspace:GetDescendants()) do
    AddPrompt(Object)
end

Workspace.DescendantAdded:Connect(function(Object)

    if Object:IsA("ProximityPrompt") then
        HarvestPrompts[Object] = true
    end
end)

Workspace.DescendantRemoving:Connect(function(Object)

    if HarvestPrompts[Object] then
        HarvestPrompts[Object] = nil
    end
end)

-- ============================================================
-- GET PROMPT PART
-- ============================================================

local function GetPromptPart(prompt)

    if not prompt or not prompt.Parent then
        return nil
    end

    if prompt.Parent:IsA("BasePart") then
        return prompt.Parent
    end

    local Model =
        prompt:FindFirstAncestorOfClass("Model")

    if Model then

        return Model.PrimaryPart
            or Model:FindFirstChildWhichIsA(
                "BasePart",
                true
            )
    end

    return nil
end

-- ============================================================
-- FIRE PROMPT
-- ============================================================

local function FireHarvest(prompt)

    if not prompt
        or not prompt.Parent
        or not prompt.Enabled then

        return false
    end

    if fireproximityprompt then

        return pcall(function()

            local OldHold =
                prompt.HoldDuration

            prompt.HoldDuration = 0

            fireproximityprompt(prompt)

            prompt.HoldDuration =
                OldHold
        end)
    end

    return pcall(function()

        prompt:InputHoldBegin()

        task.wait(0.03)

        prompt:InputHoldEnd()
    end)
end

-- ============================================================
-- HARVEST ONE
-- ============================================================

local function HarvestOne(prompt)

    if not Settings.AutoFarm then
        return
    end

    if not IsHarvestPrompt(prompt) then
        return
    end

    local Root =
        GetRoot()

    if not Root then
        return
    end

    local Part =
        GetPromptPart(prompt)

    if not Part then
        return
    end

    local Distance =
        (Root.Position - Part.Position).Magnitude

    if Distance >
        Settings.MaxFarmDistance then

        return
    end

    local ActivationDistance =
        math.max(
            prompt.MaxActivationDistance - 1,
            4
        )

    -- Bay tới cây nếu đang ngoài khoảng tương tác
    if Distance > ActivationDistance then

        Root.CFrame =
            Part.CFrame
            * CFrame.new(0,3,2)

        task.wait(0.05)
    end

    FireHarvest(prompt)

    task.wait(
        Settings.HarvestDelay
    )
end

-- ============================================================
-- AUTO FARM LOOP
-- ============================================================

task.spawn(function()

    while ScreenGui.Parent do

        if Settings.AutoFarm then

            for Prompt in pairs(
                HarvestPrompts
            ) do

                if not Settings.AutoFarm then
                    break
                end

                if Prompt
                    and Prompt.Parent then

                    pcall(
                        HarvestOne,
                        Prompt
                    )

                else

                    HarvestPrompts[Prompt] = nil

                end
            end
        end

        task.wait(
            Settings.FarmDelay
        )
    end
end)

-- ============================================================
-- DRAG UI
-- ============================================================

local Dragging = false
local DragStart
local StartPosition

Topbar.InputBegan:Connect(
    function(Input)

        if Input.UserInputType
                == Enum.UserInputType.MouseButton1
            or Input.UserInputType
                == Enum.UserInputType.Touch then

            Dragging = true
            DragStart = Input.Position
            StartPosition = Main.Position
        end
    end
)

UserInputService.InputChanged:Connect(
    function(Input)

        if Dragging
            and (
                Input.UserInputType
                    == Enum.UserInputType.MouseMovement
                or Input.UserInputType
                    == Enum.UserInputType.Touch
            ) then

            local Delta =
                Input.Position
                - DragStart

            Main.Position =
                UDim2.new(
                    StartPosition.X.Scale,
                    StartPosition.X.Offset
                        + Delta.X,

                    StartPosition.Y.Scale,
                    StartPosition.Y.Offset
                        + Delta.Y
                )
        end
    end
)

UserInputService.InputEnded:Connect(
    function(Input)

        if Input.UserInputType
                == Enum.UserInputType.MouseButton1
            or Input.UserInputType
                == Enum.UserInputType.Touch then

            Dragging = false
        end
    end
)
