local function readTelemetry()
    local data = {
        Timestamp = os.clock(),
    }

    pcall(function()
        data.MemoryMB = Stats:GetTotalMemoryUsageMb()
    end)

    pcall(function()
        data.RenderCPUms = Stats.RenderCPUFrameTime * 1000
    end)

    pcall(function()
        data.RenderGPUms = Stats.RenderGPUFrameTime * 1000
    end)

    pcall(function()
        data.DrawCalls = Stats.SceneDrawcallCount
    end)

    pcall(function()
        data.Triangles = Stats.SceneTriangleCount
    end)

    pcall(function()
        data.InstanceCount = Stats.InstanceCount
    end)

    CurrentTelemetry = data
    return data

local function numericDelta(before, after)

local function disconnectPlayerConnection(player)
    local connection = PlayerConnections[player]

    if connection then
        pcall(function()

    disconnectPlayerConnection(player)

        registerCharacter(player.Character)

    PlayerConnections[player] = player.CharacterAdded:Connect(registerCharacter)
end

local function markHumanoidModel(humanoid)
    if not humanoid then
        return
    end

    local parent = humanoid.Parent

    if parent and parent:IsA("Model") then
        HumanoidModels[parent] = true
    end
end

local function modelHasHumanoid(model)
    if HumanoidModels[model] then
        return true
    end

    local humanoid = model:FindFirstChildOfClass("Humanoid")

    if humanoid then
        HumanoidModels[model] = true
        return true
    end

    -- Không cache false vì Humanoid có thể được thêm sau.
    return false
end

local function isProtectedObject(object)
    if not object then
        return true
    end

    local className = object.ClassName

    if className == "SpawnLocation"
        or className == "Seat"
        or className == "VehicleSeat" then

        return true
    end

    local camera = Workspace.CurrentCamera
    local current = object

    while current and current ~= Workspace do
        if current == camera then
            return true
        end

        if CharacterModels[current] or HumanoidModels[current] then
            return true
        end

        local currentClass = current.ClassName

        if currentClass == "Tool" then
            return true
        end

        if currentClass == "Model" and modelHasHumanoid(current) then
            return true
        end

        current = current.Parent
    end

    return false
end

-- GROUND HEURISTIC
--
-- Chỉ có tác dụng khi HideWorld=true (POTATO).

local GroundKeywords = {
    "ground",
    "floor",
    "baseplate",
    "terrain",
    "land",
    "road",
    "path",
    "platform",
}

local function isLikelyGround(part)
    if not part or not part:IsA("BasePart") then
        return false
    end

    local lowerName = string.lower(part.Name)

    for i = 1, #GroundKeywords do
        if string.find(lowerName, GroundKeywords[i], 1, true) then
            return true
        end
    end

    local size = part.Size

    if size.Y <= 10 and size.X >= 90 and size.Z >= 90 then
        return true
    end

    return false
end

------------------------------------------------------------------------------
-- ADAPTER SYSTEM
--
-- Core không biết game cụ thể.
--
-- Adapter:
-- {
--     Name = "Example",
--     Match = function(ctx) return ctx.PlaceId == 123 end,
--     Classify = function(object, ctx)
--         return "KEEP" / "HIDE" / "DEFAULT"
--     end
-- }
------------------------------------------------------------------------------

local Adapters = {}

local function adapterMatches(adapter)
    if type(adapter) ~= "table"
        or type(adapter.Match) ~= "function"
        or type(adapter.Classify) ~= "function" then

        return false
    end

    local ok, result = pcall(adapter.Match, Context)
    return ok and result == true
end

local function addAdapter(adapter)
    if type(adapter) ~= "table" then
        return false, "adapter must be a table"
    end

    if type(adapter.Match) ~= "function"
        or type(adapter.Classify) ~= "function" then

        return false, "adapter requires Match and Classify functions"
    end

    if not adapterMatches(adapter) then
        return false, "adapter does not match this game"
    end

    Adapters[#Adapters + 1] = adapter
    table.clear(AdapterDecisionCache)

    log("Adapter enabled:", adapter.Name or tostring(#Adapters))

    return true
end

if type(ENV.AXIOM_ADAPTERS) == "table" then
    for _, adapter in ipairs(ENV.AXIOM_ADAPTERS) do
        addAdapter(adapter)
    end
end

local function classifyByAdapters(object)
    local cached = AdapterDecisionCache[object]

    if cached ~= nil then
        return cached
    end

    for i = 1, #Adapters do
        local adapter = Adapters[i]

        local ok, result = pcall(adapter.Classify, object, Context)

        if ok and type(result) == "string" then
            result = string.upper(result)

            if result == "KEEP" or result == "HIDE" then
                AdapterDecisionCache[object] = result
                return result
            end
        end
    end

    return "DEFAULT"
end

local function optimizeLightingObject(object)
    if not running or not object then
        return
    end

    if Config.DisablePostEffects and isPostEffect(object) then
        trackDynamic(object)

        pcall(function()
            object.Enabled = false
        end)

        return
    end

    if object.ClassName == "Atmosphere" then
        trackDynamic(object)

        pcall(function()
            object.Density = 0
            object.Haze = 0
            object.Glare = 0
        end)
    end
end

    optimizeLightingObject(object)

    task.defer(optimizeLightingObject, object)

-- OPTIONAL TEXTURE REFERENCE RELEASE
--
-- POTATO only by default. Client-side visual mutation; not used in AUTO/AFK.

local function releaseTextureReference(object)
    if not Config.ReleaseTextureReferences then
        return
    end

    local className = object.ClassName

    if className == "Decal" or className == "Texture" then
        pcall(function()
            object.Texture = ""
        end)

        -- Các map mới của Decal/Texture nếu tồn tại.
        pcall(function()
            object.ColorMap = ""
        end)

        pcall(function()
            object.NormalMap = ""
        end)

        pcall(function()
            object.RoughnessMap = ""
        end)

        pcall(function()
            object.MetalnessMap = ""
        end)

        return
    end

    if className == "SpecialMesh" then
        pcall(function()
            object.TextureId = ""
        end)

        return
    end

    if className == "SurfaceAppearance" then
        pcall(function()
            object.ColorMap = ""
            object.NormalMap = ""
            object.RoughnessMap = ""
            object.MetalnessMap = ""
        end)

        return
    end

    if object:IsA("MeshPart") then
        pcall(function()
            object.TextureID = ""
        end)
    end
end

------------------------------------------------------------------------------
-- VISUAL HIDE
------------------------------------------------------------------------------

local function hideVisual(object)
    if not object or not object.Parent then
        return
    end

    local className = object.ClassName

    if object:IsA("BasePart") then
        if isProtectedObject(object) or isLikelyGround(object) then
            pcall(function()
                object.CastShadow = false
                object.Reflectance = 0
            end)

            return
        end

        pcall(function()
            object.LocalTransparencyModifier = 1
            object.CastShadow = false
            object.Reflectance = 0
        end)

        releaseTextureReference(object)
        return
    end

    if className == "Decal" or className == "Texture" then
        pcall(function()
            object.LocalTransparencyModifier = 1
        end)

        releaseTextureReference(object)
        return
    end

    if className == "ParticleEmitter" then
        trackDynamic(object)

        pcall(function()
            object.Enabled = false
            object.Rate = 0
            object.LocalTransparencyModifier = 1
            object:Clear()
        end)

        if Config.ReleaseTextureReferences then
            pcall(function()
                object.Texture = ""
            end)
        end

        return
    end

    if className == "Trail" then
        trackDynamic(object)

        pcall(function()
            object.Enabled = false
            object.LocalTransparencyModifier = 1
            object:Clear()
        end)

        if Config.ReleaseTextureReferences then
            pcall(function()
                object.Texture = ""
            end)
        end

        return
    end

    if className == "Beam" then
        trackDynamic(object)

        pcall(function()
            object.Enabled = false
            object.LocalTransparencyModifier = 1
        end)

        if Config.ReleaseTextureReferences then
            pcall(function()
                object.Texture = ""
            end)
        end

        return
    end

    if isSimpleEffectClass(className) or isLightClass(className) then
        trackDynamic(object)

        pcall(function()
            object.Enabled = false
        end)

        return
    end

    if className == "BillboardGui" or className == "SurfaceGui" then
        trackDynamic(object)

        pcall(function()
            object.Enabled = false
        end)

        return
    end

    if className == "Clouds" then
        trackDynamic(object)

        pcall(function()
            object.Enabled = false
        end)

        return
    end

    if className == "SurfaceAppearance" then
        releaseTextureReference(object)
    end
end

-- CORE OPTIMIZER
------------------------------------------------------------------------------

local function optimizeObject(object, force)
    if not running or not object or not object.Parent then
        return
    end

    if not force and ProcessedOnce[object] then
        Counters.ProcessedSkipped += 1
        return
    end

    ProcessedOnce[object] = true
    Counters.Processed += 1

    local className = object.ClassName

    local adapterDecision = classifyByAdapters(object)

    if adapterDecision == "KEEP" then
        if object:IsA("BasePart") and Config.DisableShadows then
            pcall(function()
                object.CastShadow = false
            end)
        end

        return
    end

    if adapterDecision == "HIDE" then
        hideVisual(object)
        return
    end

    --------------------------------------------------------------------------
    -- AUDIO
    --------------------------------------------------------------------------

    if Config.MuteAudio
        and (className == "Sound"
            or className == "SoundGroup"
            or className == "AudioPlayer") then

        muteAudioObject(object)
        return
    end

    --------------------------------------------------------------------------
    -- PARTICLES
    --------------------------------------------------------------------------

    if Config.DisableParticles and className == "ParticleEmitter" then
        trackDynamic(object)

        pcall(function()
            object.Enabled = false
            object.Rate = 0
            object.LocalTransparencyModifier = 1
            object:Clear()
        end)

        return
    end

    if Config.DisableTrails and className == "Trail" then
        trackDynamic(object)

        pcall(function()
            object.Enabled = false
            object.LocalTransparencyModifier = 1
            object:Clear()
        end)

        return
    end

    if Config.DisableBeams and className == "Beam" then
        trackDynamic(object)

        pcall(function()
            object.Enabled = false
            object.LocalTransparencyModifier = 1
        end)

        return
    end

    if Config.DisableHighlights and className == "Highlight" then
        trackDynamic(object)

        pcall(function()
            object.Enabled = false
        end)

        return
    end

    if Config.DisableParticles
        and (className == "Smoke"
            or className == "Fire"
            or className == "Sparkles") then

        trackDynamic(object)

        pcall(function()
            object.Enabled = false
        end)

        return
    end

    --------------------------------------------------------------------------
    -- LIGHTS / CLOUDS
    --------------------------------------------------------------------------

    if Config.DisableLights and isLightClass(className) then
        trackDynamic(object)

        pcall(function()
            object.Enabled = false
        end)

        return
    end

    if className == "Clouds" then
        trackDynamic(object)

        pcall(function()
            object.Enabled = false
        end)

        return
    end

    --------------------------------------------------------------------------
    -- WORLD GUI
    --------------------------------------------------------------------------

    if className == "BillboardGui" or className == "SurfaceGui" then
        if Config.HideWorld then
            hideVisual(object)
        end

        return
    end

    --------------------------------------------------------------------------
    -- TEXTURES
    --------------------------------------------------------------------------

    if Config.HideTextures
        and (className == "Decal" or className == "Texture") then

        pcall(function()
            object.LocalTransparencyModifier = 1
        end)

        releaseTextureReference(object)
        return
    end

    if className == "SpecialMesh" then
        releaseTextureReference(object)
        return
    end

    if className == "SurfaceAppearance" then
        releaseTextureReference(object)
        return
    end

    --------------------------------------------------------------------------
    -- BASEPART
    --------------------------------------------------------------------------

    if object:IsA("BasePart") then
        local protected = isProtectedObject(object)

        pcall(function()
            if Config.DisableShadows then
                object.CastShadow = false
            end

            object.Reflectance = 0

            if Config.HideWorld
                and not protected
                and not isLikelyGround(object) then

                object.LocalTransparencyModifier = 1
            end
        end)

        if Config.HideWorld then
            releaseTextureReference(object)
        end

        return
    end
end

------------------------------------------------------------------------------
-- QUEUE
------------------------------------------------------------------------------

local function queueSize()
    if QueueTail < QueueHead then
        return 0
    end

    return QueueTail - QueueHead + 1
end

local function resetQueue()
    Queue = {}
    QueueHead = 1
    QueueTail = 0
end

local function findDirtyRoot(object)
    if not object then
        return nil
    end

    local current = object
    local parent = current.Parent

    while parent
        and parent ~= Workspace
        and parent ~= SoundService
        and parent ~= Lighting
        and parent ~= game do

        current = parent
        parent = current.Parent
    end

    if parent == Workspace
        or parent == SoundService
        or parent == Lighting then

        return current
    end

    -- Fallback an toàn.
    if object:IsDescendantOf(Workspace) then
        return Workspace
    end

    if object:IsDescendantOf(SoundService) then
        return SoundService
    end

    return nil
end

local function markDirtyRoot(object)
    local root = findDirtyRoot(object)

    if root then
        DirtyRoots[root] = true
        Counters.DirtyRootMarks += 1
    end
end

local function enqueue(object)
    if not running
        or not object
        or not object.Parent
        or not shouldProcess(object)
        or QueuedObjects[object]
        or ProcessedOnce[object] then

        return
    end

    local size = queueSize()

    if size >= Config.MaxQueueSize then
        Counters.QueueOverflow += 1
        markDirtyRoot(object)
        return
    end

    QueuedObjects[object] = true

    QueueTail += 1
    Queue[QueueTail] = object

    local newSize = size + 1

    if newSize > Counters.QueueHighWater then
        Counters.QueueHighWater = newSize
    end
end

task.spawn(function()
    while running do
        local size = queueSize()
        local batch = Config.QueueBatch

        -- Khi backlog cao, tăng nhẹ throughput nhưng không tăng vô hạn.
        if size > Config.MaxQueueSize * 0.75 then
            batch = math.floor(batch * 1.5)
        end

        local processed = 0

        while QueueHead <= QueueTail and processed < batch do
            local object = Queue[QueueHead]

            Queue[QueueHead] = nil
            QueueHead += 1
            processed += 1

            if object then
                QueuedObjects[object] = nil

                if object.Parent then
                    optimizeObject(object, false)
                end
            end
        end

        if QueueHead > QueueTail then
            resetQueue()
        end

        task.wait(Config.QueueDelay)
    end
end)

------------------------------------------------------------------------------
-- SCAN SCHEDULER
------------------------------------------------------------------------------

local function scheduleScan(root, force)
    if not running or not root or not root.Parent then
        return false
    end

    if ScanQueued[root] then
        ScanAgain[root] = true

        if force then
            ScanAgainForce[root] = true
        end

        return true
    end

    ScanQueued[root] = true

    if force then
        ScanForce[root] = true
    end

    ScanTail += 1
    ScanQueue[ScanTail] = root

    return true
end

local function resetScanQueueIfEmpty()
    if ScanHead > ScanTail then
        ScanQueue = {}
        ScanHead = 1
        ScanTail = 0
    end
end

local function scanSubtree(root, force)
    if not root or not root.Parent then
        return
    end

    local stack = { root }
    local top = 1
    local visitedThisSlice = 0

    while running and top > 0 do
        local object = stack[top]

        stack[top] = nil
        top -= 1

        if object and object.Parent then
            Counters.ScanNodesVisited += 1

            local className = object.ClassName

            if className == "Humanoid" then
                markHumanoidModel(object)
            end

            if shouldProcess(object) then
                optimizeObject(object, force)
            end

            local children = nil

            local ok = pcall(function()
                children = object:GetChildren()
            end)

            if ok and children then
                for i = 1, #children do
                    top += 1
                    stack[top] = children[i]
                end
            end

            visitedThisSlice += 1

            if visitedThisSlice >= Config.InitialNodesPerSlice then
                visitedThisSlice = 0
                task.wait(Config.InitialSliceDelay)
            end
        end
    end

    table.clear(stack)
end

task.spawn(function()
    while running do
        if ScanHead <= ScanTail then
            local root = ScanQueue[ScanHead]

            ScanQueue[ScanHead] = nil
            ScanHead += 1

            if root and root.Parent then
                local force = ScanForce[root] == true
                ScanForce[root] = nil

                scanSubtree(root, force)

                Counters.ScansCompleted += 1

                if InitialRootsPending[root] then
                    InitialRootsPending[root] = nil
                    InitialRootsRemaining -= 1

                    if InitialRootsRemaining <= 0 and not PostInitialTelemetry then
                        PostInitialTelemetry = readTelemetry()
                    end
                end
            end

            ScanQueued[root] = nil

            local scanAgain = ScanAgain[root] == true
            local scanAgainForce = ScanAgainForce[root] == true

            ScanAgain[root] = nil
            ScanAgainForce[root] = nil

            if scanAgain and root and root.Parent then
                scheduleScan(root, scanAgainForce)
            end

            resetScanQueueIfEmpty()
        else
            task.wait(0.10)
        end
    end
end)

------------------------------------------------------------------------------
-- DIRTY ROOT RECOVERY
--
-- Queue overflow không làm mất optimization vĩnh viễn.
-- Khi backlog giảm, rescan một số subtree đã bị burst.
------------------------------------------------------------------------------

task.spawn(function()
    while running do
        task.wait(Config.DirtyPassDelay)

        if queueSize() <= Config.MaxQueueSize * 0.35 then
            local scheduled = 0

            for root in pairs(DirtyRoots) do
                DirtyRoots[root] = nil

                if root and root.Parent then
                    scheduleScan(root, false)
                    scheduled += 1
                end

                if scheduled >= Config.DirtyRootsPerPass then
                    break
                end
            end
        end
    end
end)

-- Không rescan map; chỉ object có khả năng game bật lại.

local function reapplyDynamic(object)
    if not object or not object.Parent then
        DynamicObjects[object] = nil
        return
    end

    local className = object.ClassName

    if isPostEffect(object) then
        pcall(function()
            object.Enabled = false
        end)

        return
    end

    if className == "Atmosphere" then
        pcall(function()
            object.Density = 0
            object.Haze = 0
            object.Glare = 0
        end)

        return
    end

    if className == "ParticleEmitter" then
        pcall(function()
            object.Enabled = false
            object.Rate = 0
            object.LocalTransparencyModifier = 1
            object:Clear()
        end)

        return
    end

    if className == "Trail" then
        pcall(function()
            object.Enabled = false
            object.LocalTransparencyModifier = 1
            object:Clear()
        end)

        return
    end

    if className == "Beam" then
        pcall(function()
            object.Enabled = false
            object.LocalTransparencyModifier = 1
        end)

        return
    end

    if isSimpleEffectClass(className)
        or isLightClass(className)
        or className == "Clouds" then

        pcall(function()
            object.Enabled = false
        end)

        return
    end

    if className == "BillboardGui" or className == "SurfaceGui" then
        if Config.HideWorld then
            pcall(function()
                object.Enabled = false
            end)
        end

        return
    end

    if Config.MuteAudio
        and (className == "Sound"
            or className == "SoundGroup"
            or className == "AudioPlayer") then

        pcall(function()
            object.Volume = 0
        end)
    end
end

task.spawn(function()
    while running do
        task.wait(Config.DynamicRecheckSeconds)

        local processed = 0

        for object in pairs(DynamicObjects) do
            reapplyDynamic(object)

            processed += 1

            if processed >= Config.DynamicBatch then
                processed = 0
                task.wait()
            end
        end

        pcall(function()
            if Config.DisableShadows then
                Lighting.GlobalShadows = false
            end
        end)
    end
end)

ENV.AXIOM_REGISTER_ADAPTER = function(adapter)
    local ok, reason = addAdapter(adapter)

    if not ok then
        warn("[AXIOM] Adapter rejected:", reason)
        return false, reason
    end

    -- Adapter mới có thể thay đổi quyết định cho object đã xử lý.
    -- Clear weak processed cache rồi schedule một scan có force.
    table.clear(ProcessedOnce)
    table.clear(AdapterDecisionCache)

    scheduleScan(Workspace, true)

    return true
end

end

-- UNLOAD

-- Không lưu original value của hàng chục nghìn Instance vì bản thân bảng restore
-- có thể ăn RAM đáng kể. Rejoin là cách restore visuals sạch nhất.

ENV.AXIOM_UNLOAD = function()
    if not running then
        return
    end

    running = false

    for player, connection in pairs(PlayerConnections) do
        if connection then
            pcall(function()
                connection:Disconnect()
            end)
        end

        PlayerConnections[player] = nil
    end

    for i = 1, #Connections do
        local connection = Connections[i]

        if connection then
            pcall(function()
                connection:Disconnect()
            end)
        end

        Connections[i] = nil
    end

    table.clear(Queue)
    table.clear(ScanQueue)

    table.clear(CharacterModels)
    table.clear(HumanoidModels)
    table.clear(DynamicObjects)
    table.clear(ProcessedOnce)
    table.clear(QueuedObjects)
    table.clear(AdapterDecisionCache)
    table.clear(DirtyRoots)

    table.clear(ScanQueued)
    table.clear(ScanForce)
    table.clear(ScanAgain)
    table.clear(ScanAgainForce)

    table.clear(Adapters)

    ENV.__AXIOM_MULTI_GAME_V31_LOADED = nil

    print("[AXIOM] V3.1 stopped. Rejoin to restore changed visuals.")
end
