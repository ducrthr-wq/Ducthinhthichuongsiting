-- HELPERS

local function Lower(value)
    return string.lower(tostring(value or ""))
end

local function Has(text, word)
    return string.find(Lower(text), Lower(word), 1, true) ~= nil
end

local function GetCharacter()
    return Player.Character or Player.CharacterAdded:Wait()
end

local function GetRoot()
    local Character = GetCharacter()
    return Character:FindFirstChild("HumanoidRootPart")
        or Character:WaitForChild("HumanoidRootPart", 5)
end

local function GetPromptPart(prompt)
    if not prompt or not prompt.Parent then
        return nil
    end

    if prompt.Parent:IsA("BasePart") then
        return prompt.Parent
    end

    local attachment = prompt.Parent
    if attachment:IsA("Attachment") and attachment.Parent and attachment.Parent:IsA("BasePart") then
        return attachment.Parent
    end

    local model = prompt:FindFirstAncestorOfClass("Model")
    if model then
        return model.PrimaryPart
            or model:FindFirstChild("HumanoidRootPart")
            or model:FindFirstChildWhichIsA("BasePart", true)
    end

    return nil
end

local function GetPromptPosition(prompt)
    local part = GetPromptPart(prompt)
    return part and part.Position or nil
end

local function GetPromptInfo(prompt)
    local info =
        tostring(prompt.Name) .. " " ..
        tostring(prompt.ActionText) .. " " ..
        tostring(prompt.ObjectText)

    local parent = prompt.Parent

    for _ = 1, 7 do
        if not parent then
            break
        end

        info = info .. " " .. tostring(parent.Name)
        parent = parent.Parent
    end

    return Lower(info)
end

-- ============================================================
-- HARVEST DETECTION
-- ============================================================

local HarvestWords = {
    "harvestprompt",
    "harvestpart",
    "harvest",
    "collect",
    "pickup",
    "pick up",
    "pick fruit",
    "pick crop",
    "fruit",
    "crop",
}

local IgnoreHarvestWords = {
    "steal",
    "buy",
    "purchase",
    "sell",
    "trade",
    "gift",
    "shop",
    "open",
    "talk",
    "claim",
    "mail",
    "pet",
    "egg",
}

local function IsHarvestPrompt(prompt)
    if not prompt
        or not prompt.Parent
        or not prompt:IsA("ProximityPrompt")
        or not prompt.Enabled then
        return false
    end

    local info = GetPromptInfo(prompt)

    for _, word in ipairs(IgnoreHarvestWords) do
        if Has(info, word) then
            return false
        end
    end

    for _, word in ipairs(HarvestWords) do
        if Has(info, word) then
            return true
        end
    end

    return false
end

local function IsInsideOwnGarden(prompt)
    if not State.GardenAnchor then
        return false
    end

    local pos = GetPromptPosition(prompt)
    if not pos then
        return false
    end

    return (pos - State.GardenAnchor).Magnitude <= Config.GardenRadius
end

local function CanAttempt(prompt)
    local last = LastAttempt[prompt]
    if not last then
        return true
    end

    return (os.clock() - last) >= Config.PromptCooldown
end

-- ============================================================
-- FIRE PROMPT
-- ============================================================

local function FirePrompt(prompt)
    if not prompt or not prompt.Parent or not prompt.Enabled then
        return false
    end

    if fireproximityprompt then
        local ok = pcall(function()
            fireproximityprompt(prompt)
        end)

        if ok then
            return true
        end
    end

    return pcall(function()
        prompt:InputHoldBegin()
        task.wait(math.min(prompt.HoldDuration, 0.06))
        prompt:InputHoldEnd()
    end)
end

-- ============================================================
-- STABLE HARVEST QUEUE
-- ============================================================

local function GetHarvestCandidates()
    local root = GetRoot()
    if not root or not State.GardenAnchor then
        return {}
    end

    local candidates = {}

    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("ProximityPrompt")
            and IsHarvestPrompt(obj)
            and IsInsideOwnGarden(obj)
            and CanAttempt(obj) then

            local pos = GetPromptPosition(obj)

            if pos then
                table.insert(candidates, {
                    Prompt = obj,
                    Position = pos,
                    Distance = (root.Position - pos).Magnitude,
                })
            end
        end
    end

    -- Luôn xử lý trái gần nhất trước -> không bay loạn
    table.sort(candidates, function(a, b)
        return a.Distance < b.Distance
    end)

    return candidates
end

local function HarvestOne(prompt)
    if not State.AutoFarm or State.Selling then
        return
    end

    if not IsHarvestPrompt(prompt) or not IsInsideOwnGarden(prompt) then
        return
    end

    local root = GetRoot()
    local part = GetPromptPart(prompt)

    if not root or not part then
        return
    end

    LastAttempt[prompt] = os.clock()

    local activationDistance = math.max(prompt.MaxActivationDistance - 1, 4)
    local distance = (root.Position - part.Position).Magnitude

    if distance > activationDistance then
        -- Chỉ di chuyển trong vườn đã khóa
        root.CFrame = part.CFrame * CFrame.new(0, Config.TeleportHeight, 2)
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
        task.wait(0.08)
    end

    -- Retry để giảm trường hợp "trái được trái không"
    for attempt = 1, Config.HarvestRetryCount do
        if not State.AutoFarm
            or State.Selling
            or not prompt.Parent
            or not prompt.Enabled then
            break
        end

        FirePrompt(prompt)
        task.wait(Config.HarvestRetryDelay)

        -- Nếu prompt biến mất/tắt -> server đã nhận harvest
        if not prompt.Parent or not prompt.Enabled then
            break
        end
    end
end

task.spawn(function()
    while ScreenGui and ScreenGui.Parent do
        if State.AutoFarm and not State.Selling then
            local candidates = GetHarvestCandidates()
            local nearest = candidates[1]

            if nearest and nearest.Prompt then
                pcall(HarvestOne, nearest.Prompt)
            end
        end

        task.wait(Config.FarmLoopDelay)
    end
end)

local function IsSellPrompt(prompt)
    if not prompt or not prompt:IsA("ProximityPrompt") then
        return false
    end

    local info = GetPromptInfo(prompt)

    return Has(info, "steven")
        or Has(info, "sell stand")
        or Has(info, "sellstand")
        or Has(info, "sell inventory")
        or Has(info, "sell all")
        or Has(info, "sell")
end

local function FindSellPrompt()
    local root = GetRoot()
    local best = nil
    local bestPriority = -1
    local bestDistance = math.huge

    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("ProximityPrompt") and IsSellPrompt(obj) then
            local info = GetPromptInfo(obj)
            local pos = GetPromptPosition(obj)

            if pos then
                local priority = 0

                if Has(info, "steven") then
                    priority = 3
                elseif Has(info, "sell stand") or Has(info, "sellstand") then
                    priority = 2
                elseif Has(info, "sell") then
                    priority = 1
                end

                local distance = root and (root.Position - pos).Magnitude or math.huge

                if priority > bestPriority
                    or (priority == bestPriority and distance < bestDistance) then

                    best = obj
                    bestPriority = priority
                    bestDistance = distance
                end
            end
        end
    end

    return best
end

local function GetButtonText(button)
    local text = tostring(button.Name)

    if button:IsA("TextButton") then
        text = text .. " " .. tostring(button.Text)
    end

    for _, child in ipairs(button:GetDescendants()) do
        if child:IsA("TextLabel") or child:IsA("TextButton") then
            text = text .. " " .. tostring(child.Text)
        end
    end

    return Lower(text)
end

local function IsSellInventoryButton(obj)
    if not (obj:IsA("TextButton") or obj:IsA("ImageButton")) then
        return false
    end

    local text = GetButtonText(obj)

    return Has(text, "sell inventory")
        or Has(text, "sell all")
        or Has(text, "sell everything")
        or Has(text, "sell crops")
end

local function FindSellInventoryButton()
    for _, obj in ipairs(PlayerGui:GetDescendants()) do
        if IsSellInventoryButton(obj) then
            return obj
        end
    end

    return nil
end

local function PressGuiButton(button)
    if not button or not button.Parent then
        return false
    end

    if firesignal then
        local ok = pcall(function()
            firesignal(button.Activated)
        end)

        if ok then
            return true
        end
    end

    local ok = pcall(function()
        button:Activate()
    end)

    if ok then
        return true
    end

    if button.AbsoluteSize.X > 0 and button.AbsoluteSize.Y > 0 then
        local x = button.AbsolutePosition.X + button.AbsoluteSize.X / 2
        local y = button.AbsolutePosition.Y + button.AbsoluteSize.Y / 2

        return pcall(function()
            VirtualInputManager:SendMouseButtonEvent(x, y, 0, true, game, 0)
            task.wait(0.05)
            VirtualInputManager:SendMouseButtonEvent(x, y, 0, false, game, 0)
        end)
    end

    return false
end

local function WaitForSellButton(timeout)
    local started = os.clock()

    repeat
        local button = FindSellInventoryButton()
        if button then
            return button
        end

        task.wait(0.10)
    until os.clock() - started >= timeout

    return nil
end

local function SellAll()
    if State.Selling then
        return false
    end

    State.Selling = true

    local root = GetRoot()

    if not root then
        State.Selling = false
        return false
    end

    local returnCFrame = root.CFrame
    local success = false

    local ok = pcall(function()
        -- Nếu menu sell đang mở rồi thì bấm luôn
        local existingButton = FindSellInventoryButton()

        if existingButton then
            success = PressGuiButton(existingButton)
            task.wait(0.7)
            return
        end

        local sellPrompt = FindSellPrompt()

        if not sellPrompt then
            warn("[AXIOM] Không tìm thấy Steven / Sell Stand")
            return
        end

        local sellPart = GetPromptPart(sellPrompt)

        if not sellPart then
            warn("[AXIOM] Không tìm được vị trí Sell Stand")
            return
        end

        root.CFrame = sellPart.CFrame * CFrame.new(0, 3, 3)
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
        task.wait(0.45)

        FirePrompt(sellPrompt)

        local button = WaitForSellButton(Config.SellMenuTimeout)

        if not button then
            -- Retry mở menu lần 2
            FirePrompt(sellPrompt)
            button = WaitForSellButton(2.5)
        end

        if button then
            success = PressGuiButton(button)
            task.wait(0.8)
        else
            warn("[AXIOM] Đã tới điểm bán nhưng không thấy nút Sell Inventory")
        end
    end)

    if root and root.Parent and returnCFrame then
        root.CFrame = returnCFrame
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
    end

    State.Selling = false

    if not ok then
        warn("[AXIOM] Sell loop gặp lỗi")
    end

    return success
end

task.spawn(function()
    while task.wait(Config.SellInterval) do
        if not ScreenGui or not ScreenGui.Parent then
            break
        end

        if State.AutoSell and not State.Selling then
            pcall(SellAll)
        end
    end
end)

getgenv().AXIOM_SellNow = function()
    task.spawn(SellAll)
end

local function SwitchPage(pageName, button)
    for _, page in pairs(Pages) do
        page.Visible = false
    end

    Pages[pageName].Visible = true

    for _, object in ipairs(Sidebar:GetChildren()) do
        if object:IsA("TextButton") then
            object.BackgroundColor3 = Color3.fromRGB(28, 31, 41)
            object.TextColor3 = Color3.fromRGB(175, 180, 195)
        end
    end

    button.BackgroundColor3 = Color3.fromRGB(38, 84, 150)
    button.TextColor3 = Color3.new(1, 1, 1)
end

local function CreateTab(text, pageName)
    local Button = Instance.new("TextButton")
    Button.Size = UDim2.new(1, 0, 0, 42)
    Button.BackgroundColor3 = Color3.fromRGB(28, 31, 41)
    Button.BorderSizePixel = 0
    Button.Text = "  " .. text
    Button.TextColor3 = Color3.fromRGB(175, 180, 195)
    Button.TextSize = 13
    Button.Font = Enum.Font.GothamMedium
    Button.TextXAlignment = Enum.TextXAlignment.Left
    Button.AutoButtonColor = false
    Button.Parent = Sidebar

    local Corner = Instance.new("UICorner")
    Corner.CornerRadius = UDim.new(0, 8)
    Corner.Parent = Button

    Button.MouseButton1Click:Connect(function()
        SwitchPage(pageName, Button)
    end)

    return Button
end

local Open = true

local function SetMenu(state)
    Open = state

    if Open then
        Main.Visible = true
        Main.Size = UDim2.fromOffset(520, 330)

        TweenService:Create(
            Main,
            TweenInfo.new(0.18, Enum.EasingStyle.Quart),
            {Size = UDim2.fromOffset(560, 360)}
        ):Play()
    else
        Main.Visible = false
    end
end

FloatButton.MouseButton1Click:Connect(function()
    SetMenu(not Open)
end)

Close.MouseButton1Click:Connect(function()
    SetMenu(false)
end)

-- ============================================================
-- DRAG UI
-- ============================================================

local Dragging = false
local DragStart
local StartPosition

Topbar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        Dragging = true
        DragStart = input.Position
        StartPosition = Main.Position
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if Dragging
        and (
            input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch
        ) then

        local Delta = input.Position - DragStart

        Main.Position = UDim2.new(
            StartPosition.X.Scale,
            StartPosition.X.Offset + Delta.X,
            StartPosition.Y.Scale,
            StartPosition.Y.Offset + Delta.Y
        )
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        Dragging = false
    end
end)
